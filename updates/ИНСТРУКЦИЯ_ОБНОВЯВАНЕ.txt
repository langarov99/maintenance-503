================================================================
  ИНСТРУКЦИЯ ЗА ОБНОВЯВАНЕ НА DATA EXTRACTION BOT
================================================================


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  СТЪПКА 1 — ИЗТЕГЛЯНЕ НА ПРОМЕНИТЕ (MASTER машина — F:\)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Отворете Command Prompt (cmd) — Start → напишете "cmd" → Enter

2. Отидете в папката на бота:
     cd F:\maintenance-503

3. Изтеглете последните промени от сървъра:
     git pull origin claude/data-extraction-bot-Gpvpe


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  СТЪПКА 2 — ГЕНЕРИРАНЕ НА PATCH (MASTER машина — F:\)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Щракнете двукратно върху:
     F:\maintenance-503\updates\GENERATE_UPDATE.bat

2. Програмата ще покаже текущата и новата версия.
   Въведете брой commit-и за включване (обикновено: 1) → Enter

3. Създава се файл:
     F:\maintenance-503\updates\update_X.Y.patch


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  СТЪПКА 3 — ИЗПРАЩАНЕ НА PATCH ФАЙЛА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Изпратете файла update_X.Y.patch на другите машини чрез:
  - Имейл
  - USB стик
  - Viber / WhatsApp
  - Всеки друг удобен начин


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  СТЪПКА 4 — ПРИЛАГАНЕ НА PATCH (другите машини)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Уверете се, че ботът НЕ работи
   (затворете прозореца на START.bat ако е пуснат)

2. Поставете update_X.Y.patch в папката:
     [флашка]\maintenance-503\updates\

3. Щракнете двукратно върху:
     [флашка]\maintenance-503\updates\APPLY_UPDATE.bat

4. Изчакайте съобщението:
     [OK] Обновлението е приложено успешно!
          Версия: X.Y --> X.Z

5. Стартирайте бота с START.bat


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ВАЖНО
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

- Прилагайте обновленията по ред — не прескачайте версии!
  Пример: 1.0 → 1.1 → 1.2 (не от 1.0 директно на 1.2)

- Patch файлът се изтрива автоматично след успешно прилагане.

- Текущата версия на всяка машина се вижда във файла:
    updates\version.txt

- Файловете с данни (*.xlsx в data\suppliers\) НЕ се включват
  в patch-а — копирайте ги ръчно при нужда.
