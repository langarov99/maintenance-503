━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  СТЪПКА 1 — ГЕНЕРИРАНЕ НА PATCH (прави се на MASTER машината)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

След като разработчикът направи промени и ги качи в GitHub,
отворете Command Prompt (cmd) в папката на бота и изпълнете:

  git pull origin claude/data-extraction-bot-Gpvpe

След това генерирайте patch файл с командата:

  git format-patch HEAD~1 --stdout > updates\update.patch

  (Ако промените са повече от 1 commit, сменете ~1 с броя commits,
   например за 3 промени: git format-patch HEAD~3 --stdout > updates\update.patch)

Файлът update.patch ще се създаде в папката updates\.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  СТЪПКА 2 — ИЗПРАЩАНЕ НА PATCH ФАЙЛА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Изпратете файла updates\update.patch на другите машини чрез:
  - Имейл
  - USB стик
  - Viber / WhatsApp
  - Всеки друг удобен начин


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  СТЪПКА 3 — ПРИЛАГАНЕ НА PATCH (на другите машини)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Уверете се, че ботът НЕ работи (затворете START.bat ако е пуснат)

2. Поставете update.patch в папката updates\ на бота

3. Щракнете двукратно върху updates\APPLY_UPDATE.bat

4. Изчакайте съобщението "[OK] Обновлението е приложено успешно!"

5. Стартирайте бота с START.bat


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ВАЖНО
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

- Patch файловете трябва да се прилагат по ред!
  Не прескачайте обновления.

- Ако видите грешка "вече е приложен" — обновлението е OK,
  не е нужно да правите нищо.

- Файловете с данни (*.xlsx в data\suppliers\) НЕ се включват
  в patch-а и трябва да се копират ръчно при нужда.
